```
@given(st.lists(st.integers(0, 100), min_size=5, max_size=10))

    verbosity=Verbosity.debug,
    max_examples=100,
    phases=[Phase.generate, Phase.shrink],
    database=None,
)
def test_node_deletion(lst):
    """
    触发条件：当列表长度大于7且前三个元素之和大于50时失败
    约简目标：删除多余的列表元素，同时保持前三个元素之和>50
    """
    print(f"测试输入: {lst}, 长度: {len(lst)}, 前3项和: {sum(lst[:3])}")
    if len(lst) > 7 and sum(lst[:3]) > 50:
        print("触发失败条件！")
        assert False

if __name__ == "__main__":
    test_node_deletion()
```

在本例中，shrinker需要删除列表末尾的多余元素（索引3及之后的元素），但不能删除前3个关键元素，因为它们的和必须保持>50。

```
def run_node_program(self, i, program, original, repeats=1):
    if i + len(program) > len(original.nodes) or i < 0:
        return False
    attempt = list(original.nodes)
    for _ in range(repeats):

            j = i + k
```
