## 概述
前面的几份笔记分别讨论了 [[Harness Engineering/02_运行时循环与状态机#概述|运行时循环]]、[[Harness Engineering/03_上下文工程(Context Engineering)#概述|上下文工程]]、[[Harness Engineering/04_工具注册与执行层#概述|工具注册与执行层]]、[[Harness Engineering/05_可靠性与安全#概述|可靠性与安全]] 和 [[Harness Engineering/06_评测与可观测性#概述|评测与可观测性]]。这一篇的目标是把这些部件拼成一个最小可用的端到端 harness。

这里的“最小可用”并不追求最少代码，而是满足四个条件：
- 能完成 `user -> model -> tool -> model -> answer` 的完整闭环。
- 有明确的状态对象和预算控制。
- 有工具参数校验与统一 observation 格式。
- 有基础日志和错误路径。

## 架构目标
最小实现至少应包含以下组件：
- `ModelClient`：负责与模型 API 通信。
- `ToolRegistry`：负责工具 Schema 导出、参数校验和执行。
- `ContextAssembler`：负责把系统提示、历史消息和工具规格组织为输入。
- `AgentHarness`：负责驱动主循环、跟踪状态和预算。

这个结构可以视为：
$$Harness = Runtime + Context + Tools + Policies$$

## 完整示例代码
下面给出一个单文件版本的最小实现。它使用 OpenAI 官方 Python SDK、`pydantic v2` 做参数校验，并给出一个天气查询工具作为演示。

```python
from __future__ import annotations

import json
import logging
import os
import time
import uuid
from dataclasses import dataclass, field
from typing import Any

from openai import OpenAI
from pydantic import BaseModel, Field, ValidationError


logging.basicConfig(level=logging.INFO)
logger = logging.getLogger("mini_harness")


class WeatherInput(BaseModel):
    location: str = Field(min_length=1, max_length=100, description="城市名称")
    unit: str = Field(default="celsius", pattern="^(celsius|fahrenheit)$")


def get_weather(params: WeatherInput) -> dict[str, Any]:
    # 真实系统中，这里应调用外部 API，并配置 timeout / retry / allowlist
    mock_data = {
        "Beijing": {"temperature": 28, "condition": "Sunny"},
        "Shanghai": {"temperature": 25, "condition": "Cloudy"},
        "London": {"temperature": 17, "condition": "Rainy"},
    }
    result = mock_data.get(
        params.location,
        {"temperature": 20, "condition": "Unknown"},
    )
    return {
        "location": params.location,
        "unit": params.unit,
        "temperature": result["temperature"],
        "condition": result["condition"],
    }


@dataclass(slots=True)
class ToolSpec:
    name: str
    description: str
    input_model: type[BaseModel]
    handler: Any
    risk_level: str = "low"


class ToolRegistry:
    def __init__(self) -> None:
        self._tools: dict[str, ToolSpec] = {}

    def register(self, spec: ToolSpec) -> None:
        if spec.name in self._tools:
            raise ValueError(f"duplicate tool: {spec.name}")
        self._tools[spec.name] = spec

    def openai_schemas(self) -> list[dict[str, Any]]:
        schemas: list[dict[str, Any]] = []
        for spec in self._tools.values():
            schemas.append(
                {
                    "type": "function",
                    "function": {
                        "name": spec.name,
                        "description": spec.description,
                        "parameters": spec.input_model.model_json_schema(),
                    },
                }
            )
        return schemas

    def execute(self, tool_name: str, raw_arguments: str) -> str:
        if tool_name not in self._tools:
            return json.dumps({"ok": False, "error": f"unknown_tool: {tool_name}"}, ensure_ascii=False)

        spec = self._tools[tool_name]

        try:
            arguments = json.loads(raw_arguments)
        except json.JSONDecodeError as exc:
            return json.dumps({"ok": False, "error": f"invalid_json: {exc}"}, ensure_ascii=False)

        try:
            validated = spec.input_model.model_validate(arguments)
        except ValidationError as exc:
            return json.dumps(
                {"ok": False, "error": "validation_error", "detail": exc.errors()},
                ensure_ascii=False,
            )

        try:
            result = spec.handler(validated)
        except Exception as exc:
            logger.exception("tool_execution_failed")
            return json.dumps(
                {"ok": False, "error": f"tool_execution_failed: {exc}"},
                ensure_ascii=False,
            )

        return json.dumps({"ok": True, "result": result}, ensure_ascii=False)


class ContextAssembler:
    def __init__(self, system_prompt: str) -> None:
        self.system_prompt = system_prompt

    def build_messages(self, history: list[dict[str, Any]]) -> list[dict[str, Any]]:
        return [{"role": "system", "content": self.system_prompt}] + history


class OpenAIModelClient:
    def __init__(self, model: str) -> None:
        self.client = OpenAI(api_key=os.environ["OPENAI_API_KEY"])
        self.model = model

    def create(
        self,
        messages: list[dict[str, Any]],
        tools: list[dict[str, Any]],
    ):
        return self.client.chat.completions.create(
            model=self.model,
            messages=messages,
            tools=tools,
            tool_choice="auto",
            temperature=0,
        )


@dataclass(slots=True)
class RuntimeBudget:
    max_steps: int = 6
    max_tool_calls: int = 6
    max_elapsed_seconds: float = 30.0


@dataclass(slots=True)
class AgentState:
    run_id: str
    history: list[dict[str, Any]]
    step_index: int = 0
    tool_calls: int = 0
    final_answer: str | None = None
    error: str | None = None
    started_at: float = field(default_factory=time.monotonic)


class AgentHarness:
    def __init__(
        self,
        model_client: OpenAIModelClient,
        tool_registry: ToolRegistry,
        context_assembler: ContextAssembler,
        budget: RuntimeBudget | None = None,
    ) -> None:
        self.model_client = model_client
        self.tool_registry = tool_registry
        self.context_assembler = context_assembler
        self.budget = budget or RuntimeBudget()

    def _check_budget(self, state: AgentState) -> None:
        if state.step_index >= self.budget.max_steps:
            raise RuntimeError("max_steps_exceeded")
        if state.tool_calls >= self.budget.max_tool_calls:
            raise RuntimeError("max_tool_calls_exceeded")
        if time.monotonic() - state.started_at >= self.budget.max_elapsed_seconds:
            raise RuntimeError("max_elapsed_seconds_exceeded")

    def run(self, user_input: str) -> AgentState:
        state = AgentState(
            run_id=str(uuid.uuid4()),
            history=[{"role": "user", "content": user_input}],
        )

        while state.final_answer is None and state.error is None:
            try:
                self._check_budget(state)
            except Exception as exc:
                state.error = str(exc)
                break

            state.step_index += 1
            logger.info("run_id=%s step=%s", state.run_id, state.step_index)

            messages = self.context_assembler.build_messages(state.history)

            try:
                response = self.model_client.create(
                    messages=messages,
                    tools=self.tool_registry.openai_schemas(),
                )
            except Exception as exc:
                logger.exception("model_call_failed")
                state.error = f"model_call_failed: {exc}"
                break

            message = response.choices[0].message

            if not message.tool_calls:
                state.final_answer = message.content or ""
                state.history.append({"role": "assistant", "content": state.final_answer})
                break

            state.history.append(message.model_dump(exclude_none=True))

            for tool_call in message.tool_calls:
                state.tool_calls += 1
                observation = self.tool_registry.execute(
                    tool_name=tool_call.function.name,
                    raw_arguments=tool_call.function.arguments,
                )
                state.history.append(
                    {
                        "role": "tool",
                        "tool_call_id": tool_call.id,
                        "name": tool_call.function.name,
                        "content": observation,
                    }
                )

        return state


def build_harness() -> AgentHarness:
    registry = ToolRegistry()
    registry.register(
        ToolSpec(
            name="get_weather",
            description="获取指定城市的当前天气信息。",
            input_model=WeatherInput,
            handler=get_weather,
        )
    )

    system_prompt = (
        "你是一个严谨的智能体。"
        "当问题需要实时外部信息时，优先使用工具。"
        "如果工具返回错误，基于 observation 解释失败原因，不要编造结果。"
    )

    return AgentHarness(
        model_client=OpenAIModelClient(model="gpt-4o-mini"),
        tool_registry=registry,
        context_assembler=ContextAssembler(system_prompt=system_prompt),
        budget=RuntimeBudget(max_steps=6, max_tool_calls=4, max_elapsed_seconds=20.0),
    )


if __name__ == "__main__":
    harness = build_harness()
    state = harness.run("请告诉我 London 当前天气，并简要解释适合穿什么。")
    print("run_id:", state.run_id)
    print("final_answer:", state.final_answer)
    print("error:", state.error)
```

## 代码结构解读

### 1. `ToolRegistry`
负责三件事：
- 向模型导出工具 Schema。
- 解析并校验模型给出的参数。
- 把执行结果标准化成 JSON observation。

### 2. `ContextAssembler`
这个示例只做了最简系统提示拼接。真实系统中应继续加入：
- 最近历史消息截断。
- 检索记忆注入。
- 风险策略注入。
- 输出契约注入。

相关设计可回看 [[Harness Engineering/03_上下文工程(Context Engineering)#常见上下文层次|上下文层次]]。

### 3. `AgentHarness`
负责任务级循环，而不是单次模型请求。它显式维护：
- 步数预算。
- 工具调用预算。
- 总耗时预算。
- 最终答案与错误状态。

## 为什么这是“最小可用”
这个实现已经具备了真实 harness 的骨架：
- 不是把所有逻辑写进一个 `ask_llm()` 函数。
- 能完整处理多轮 tool call。
- 工具参数有校验，而不是直接 `json.loads()` 后执行。
- 错误不会静默吞掉，而会写入 `state.error`。
